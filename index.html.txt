        // MÁSCARA AUTOMÁTICA DE TELEFONE (DDD) 9XXXX-XXXX
        function maskPhone(input) {
            let value = input.value.replace(/\D/g, '');
            if (value.length > 11) value = value.slice(0, 11);
            
            if (value.length > 6) {
                value = `(${value.slice(0, 2)}) ${value.slice(2, 7)}-${value.slice(7)}`;
            } else if (value.length > 2) {
                value = `(${value.slice(0, 2)}) ${value.slice(2)}`;
            } else if (value.length > 0) {
                value = `(${value}`;
            }
            
            input.value = value;
        }

        function handleRegister(e) {
            e.preventDefault();
            clearErrorMsgs();

            const name = document.getElementById('regName').value.trim();
            const phone = document.getElementById('regPhone').value.trim();
            const email = document.getElementById('regEmail').value.trim().toLowerCase();
            const password = document.getElementById('regPassword').value;
            const confirmPassword = document.getElementById('regConfirmPassword').value;

            if (password !== confirmPassword) {
                triggerShakeError('registerBox', 'registerErrorMsg', 'registerErrorText', 'As senhas não coincidem!');
                return;
            }

            const cleanPhone = phone.replace(/\D/g, '');
            if (cleanPhone.length < 10) {
                triggerShakeError('registerBox', 'registerErrorMsg', 'registerErrorText', 'Informe um telefone válido com DDD!');
                return;
            }

            const emailExists = users.find(u => u.email === email);
            if (emailExists) {
                triggerShakeError('registerBox', 'registerErrorMsg', 'registerErrorText', 'Este e-mail já está cadastrado!');
                return;
            }

            const phoneExists = users.find(u => u.phone.replace(/\D/g, '') === cleanPhone);
            if (phoneExists) {
                triggerShakeError('registerBox', 'registerErrorMsg', 'registerErrorText', 'Este WhatsApp já está cadastrado!');
                return;
            }

            const newUser = { id: 'USR-' + Date.now(), name, phone, email, password, isAdmin: false };
            users.push(newUser);
            localStorage.setItem('rifa_users', JSON.stringify(users));

            currentUser = newUser;
            localStorage.setItem('rifa_current_user', JSON.stringify(currentUser));
            
            clearAuthForms();
            checkUserSession();
            alert('Cadastro realizado com sucesso!');
        }

        function cleanExpiredReservations() {
            const now = new Date().getTime();
            const twentyFourHoursMs = 24 * 60 * 60 * 1000;
            let hasChanges = false;

            numberReservations = numberReservations.filter(res => {
                if (res.status === 'pending') {
                    const reservationTime = new Date(res.createdAt).getTime();
                    if (now - reservationTime > twentyFourHoursMs) {
                        hasChanges = true;
                        return false;
                    }
                }
                return true;
            });

            if (hasChanges) {
                saveReservations();
                renderGrid();
            }
        }

        function renderGrid() {
            const gridContainer = document.getElementById('numbersGrid');
            if(!gridContainer) return;
            gridContainer.innerHTML = '';

            const numberStatusMap = {};
            numberReservations.forEach(res => {
                numberStatusMap[res.number] = res.status;
            });

            for (let i = 1; i <= appConfig.totalNumbers; i++) {
                const btn = document.createElement('button');
                btn.type = 'button';
                btn.className = 'number-btn';
                btn.textContent = i;

                const status = numberStatusMap[i];

                if (status === 'confirmed') {
                    btn.classList.add('confirmed');
                    btn.title = 'Número Vendido / Confirmado';
                } else if (status === 'pending') {
                    btn.classList.add('reserved');
                    btn.title = 'Reservado (Aguardando Aprovação de Pagamento)';
                } else {
                    if (selectedNumbers.includes(i)) {
                        btn.classList.add('selected');
                    }
                    btn.onclick = () => toggleNumberSelection(i);
                }

                gridContainer.appendChild(btn);
            }

            updateCheckoutBar();
        }

        function toggleNumberSelection(num) {
            const index = selectedNumbers.indexOf(num);
            if (index > -1) {
                selectedNumbers.splice(index, 1);
            } else {
                selectedNumbers.push(num);
            }
            selectedNumbers.sort((a, b) => a - b);
            renderGrid();
        }

        // COPIAR CHAVE PIX COM FALLBACK COMPATÍVEL
        function copyPixKey() {
            const pixKey = appConfig.pixKey;
            const btnLabel = document.getElementById('copyBtnLabel');

            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(pixKey).then(() => {
                    showCopiedSuccess(btnLabel);
                }).catch(() => {
                    fallbackCopyText(pixKey, btnLabel);
                });
            } else {
                fallbackCopyText(pixKey, btnLabel);
            }
        }

        function fallbackCopyText(text, btnLabel) {
            const textArea = document.createElement("textarea");
            textArea.value = text;
            textArea.style.position = "fixed";
            textArea.style.opacity = "0";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();
            try {
                document.execCommand('copy');
                showCopiedSuccess(btnLabel);
            } catch (err) {
                alert('Chave PIX: ' + text);
            }
            document.body.removeChild(textArea);
        }

        }

        function saveUserEditsByAdmin(e) {
            e.preventDefault();
            const id = document.getElementById('editUserId').value;
            const name = document.getElementById('editUserName').value.trim();
            const phone = document.getElementById('editUserPhone').value.trim();
            const email = document.getElementById('editUserEmail').value.trim().toLowerCase();
            const password = document.getElementById('editUserPassword').value;

            const user = users.find(u => u.id === id);
            if (user) {
                user.name = name;
                user.phone = phone;
                user.email = email;
                user.password = password;

                localStorage.setItem('rifa_users', JSON.stringify(users));

                // Atualizar também nos registros das reservas vinculadas
                numberReservations.forEach(r => {
                    if (r.userId === id) {
                        r.customerName = name;
                        r.customerPhone = phone;
                        r.customerEmail = email;
                    }
                });
                saveReservations();

                closeModal('editUserModal');
                renderAdminDashboard();
                alert('Dados do usuário atualizados com sucesso!');
            }
        }

        function approveSingleNumber(id) {
            const res = numberReservations.find(r => r.id === id);
            if (res) {
                res.status = 'confirmed';
                saveReservations();
                renderGrid();
                renderAdminDashboard();
            }
        }

        function deleteSingleNumber(id) {
            if (confirm('Tem certeza que deseja cancelar a reserva deste número específico? Ele voltará a ficar disponível.')) {
                numberReservations = numberReservations.filter(r => r.id !== id);
                saveReservations();
                renderGrid();
                renderAdminDashboard();
            }
        }

        function saveAdminConfigs() {
            const price = parseFloat(document.getElementById('cfgPrice').value);
            const phone = document.getElementById('cfgPhone').value.trim();
            const pix = document.getElementById('cfgPix').value.trim();

            if (price > 0) appConfig.pricePerNumber = price;
            if (phone) appConfig.whatsappPhone = phone;
            if (pix) appConfig.pixKey = pix;

            localStorage.setItem('rifa_config', JSON.stringify(appConfig));
            updateHeaderInfo();
            renderGrid();
            alert('Configurações salvas!');
        }

        function openModal(id) {
            if (id === 'myNumbersModal') {
                searchCustomerTickets();
            }
            document.getElementById(id).classList.remove('hidden');
        }

        function closeModal(id) {
            document.getElementById(id).classList.add('hidden');
        }

        function saveReservations() {
            localStorage.setItem('rifa_number_reservations', JSON.stringify(numberReservations));
        }
